# M0 — технические проверки: план исполнения

> **Для исполнителя-агента:** ОБЯЗАТЕЛЬНЫЙ СКИЛЛ: `superpowers:subagent-driven-development`
> (рекомендуется) либо `superpowers:executing-plans`. Шаги отмечаются флажками (`- [ ]`).

**Цель:** закрыть шесть открытых вопросов SPEC §15 измеримыми ответами до того, как под них будет
написан код демона, лаунчера и гейта.

**Подход:** каждая проба — отдельный скрипт в `tools/probes/`, который выводит факты, и отчёт в
`docs/probes/` с ответом и следствиями. Пробы не производственный код: в пакет не входят, тестами
не покрываются. Если ответ меняет принятое решение — новый ADR (нумерация с 0015) или поправка
существующего, затем правка соответствующего раздела `SPEC.md`.

**Стек:** Python 3.12 под `uv`, SDK `mcp`, `httpx`, `lxml`, `ahocorasick_rs`, `snowballstemmer`.

**Спецификация:** [SPEC.md](../../SPEC.md) раздел 15 «Технические проверки»; термины —
[CONTEXT.md](../../CONTEXT.md).

## Общие ограничения

- Python ≥ 3.12; целевая ОС — Windows; запуск пакета через `uvx odata1c …` (SPEC §11.1).
- Состав зависимостей задан SPEC §11.1: `mcp`, `httpx`, `pydantic`, `pyyaml`, `lxml`,
  `snowballstemmer`, `ahocorasick_rs`, `uvicorn`, необязательно `keyring`. Замена любой из них —
  повод для ADR, а не решение по ходу.
- Демон слушает только `127.0.0.1`, порт по умолчанию 7171 (SPEC §2.2, §3.3). Пробы, поднимающие
  сервер, берут порт 7179, чтобы не конфликтовать с рабочим демоном.
- Язык отчётов — русский, терминология по `CONTEXT.md`, запрещённые синонимы (`_Avoid_`) не
  использовать: демон, лаунчер, гейт, страж, словарь, токен, класс, база, сущность.
- Каталог `tools/probes/` в поставку PyPI не включается.
- Ни одна проба не обращается к боевой базе. Проба P4 идёт только на тестовой базе 1С:
  `https://1c.example.local/TradeDev/odata/standard.odata/` (публикация проверена
  2026-09-08: HTTPS отвечает `401`, порт 80 на этом пути не отвечает).
- Файлы `.ps1` с кириллицей сохранять в UTF-8 **с BOM**: Windows PowerShell 5.1 без BOM читает их
  как ANSI и ломается на первом же комментарии (находка пробы P1). На `.py` не распространяется.

## Шаблон отчёта

Отчёт `docs/probes/PNN-<слаг>.md` состоит из пяти разделов: **Вопрос** (цитата пункта SPEC §15),

команды, факты), **Ответ** (одно предложение), **Следствия** (что меняется в SPEC/ADR или «ничего»).

```python
"""Проба P2 (SPEC §15.1): доходит ли _meta тула до клиента через FastMCP.

Печатает три вещи: что SDK принимает при объявлении тула, какие поля есть у типа Tool,
что клиент видит в ответе tools/list.
"""
import asyncio
import inspect
import json

import mcp.types as types
from mcp.server.fastmcp import FastMCP

META = {"anthropic/requiresUserInteraction": True, "anthropic/maxResultSizeChars": 120000}


def show_api() -> None:
    print("--- сигнатура FastMCP.tool ---")
    print(inspect.signature(FastMCP.tool))
    print("--- поля types.Tool ---")
    print(sorted(types.Tool.model_fields))
    print("--- псевдонимы полей types.Tool ---")
    print({n: f.alias for n, f in types.Tool.model_fields.items() if f.alias})


def build_server() -> tuple[FastMCP, str]:
    """Собрать сервер с одним тулом; вернуть сервер и описание использованного способа."""
    server = FastMCP("odata1c-probe")
    try:
        decorator = server.tool(name="odata1c_commit", meta=META)
        way = "аргумент meta= у декоратора tool()"
    except TypeError as exc:
        print(f"meta= у декоратора не принят: {exc}")
        decorator = server.tool(name="odata1c_commit")
        way = "meta= недоступен, пробуется правка объекта Tool после регистрации"

    @decorator
    def commit(pending_id: str) -> str:
        """Выполнить подготовленную операцию записи."""
        return pending_id

    print(f"--- способ объявления: {way} ---")
    return server, way


def patch_after_registration(server: FastMCP) -> None:
    """Запасной путь: дописать _meta в уже зарегистрированный тул."""
    for tool in server._tool_manager.list_tools():
        print("объект тула:", type(tool).__name__, sorted(getattr(tool, "model_fields", {})))
        if hasattr(tool, "meta"):
            tool.meta = META
            print("meta проставлен правкой объекта (приватный API)")
        else:
            print("у объекта тула поля meta нет")


async def list_via_client(server: FastMCP) -> None:
    print("--- что видит клиент ---")
    from mcp.shared.memory import create_connected_server_and_client_session

    async with create_connected_server_and_client_session(server._mcp_server) as client:
        result = await client.list_tools()
        print(json.dumps(result.model_dump(by_alias=True, exclude_none=True),
                         ensure_ascii=False, indent=2))


async def main() -> None:
    show_api()
    server, _ = build_server()
    patch_after_registration(server)
    await list_via_client(server)


if __name__ == "__main__":
    asyncio.run(main())
```

### Задача 3: P3 — elicitation сквозь лаунчер

Отвечает на SPEC §15 пункт 2. Лаунчер обязан пробрасывать запросы сервера к клиенту в обратную

**Интерфейсы:**

- Отдаёт: подтверждение или опровержение схемы проксирования для лаунчера в плане M1d; при успехе —

```python
"""Проба P3: демон-заглушка со Streamable HTTP и тулом, который спрашивает пользователя."""
from pydantic import BaseModel, Field

from mcp.server.fastmcp import Context, FastMCP

server = FastMCP("odata1c-probe-daemon", host="127.0.0.1", port=7179)


class Confirm(BaseModel):
    approve: bool = Field(description="Подтвердить запись в 1С")


@server.tool(name="odata1c_probe_commit")
async def probe_commit(pending_id: str, ctx: Context) -> str:
    """Спросить подтверждение и вернуть ответ пользователя."""
    result = await ctx.elicit(message=f"Выполнить операцию {pending_id}?", schema=Confirm)
    return f"action={result.action} data={getattr(result, 'data', None)}"


if __name__ == "__main__":
    server.run(transport="streamable-http")
```

```python
"""Проба P3: лаунчер — stdio-клиент снаружи, Streamable HTTP-клиент внутрь, без своей логики.

Задача пробы: убедиться, что запрос сервера к клиенту (elicitation/create) доходит до stdio-клиента
и ответ возвращается обратно.
"""
import anyio
import mcp.types as types
from mcp.client.session import ClientSession
from mcp.client.streamable_http import streamablehttp_client
from mcp.server.lowlevel import Server
from mcp.server.stdio import stdio_server

URL = "http://127.0.0.1:7179/mcp"


async def main() -> None:
    async with streamablehttp_client(URL) as (up_read, up_write, _):
        async def on_elicit(context, params: types.ElicitRequestParams):
            """Запрос демона к пользователю: переадресовать своему stdio-клиенту."""
            return await proxy.request_context.session.send_request(
                types.ServerRequest(types.ElicitRequest(method="elicitation/create", params=params)),
                types.ElicitResult,
            )

        async with ClientSession(up_read, up_write, elicitation_callback=on_elicit) as upstream:
            init = await upstream.initialize()
            proxy = Server(init.serverInfo.name)

            @proxy.list_tools()
            async def list_tools() -> list[types.Tool]:
                return (await upstream.list_tools()).tools

            @proxy.call_tool()
            async def call_tool(name: str, arguments: dict) -> types.CallToolResult:
                return await upstream.call_tool(name, arguments)

            async with stdio_server() as (read, write):
                await proxy.run(read, write, proxy.create_initialization_options())


if __name__ == "__main__":
    anyio.run(main)
```

```python
"""Проба P3: stdio-клиент, отвечающий на elicitation вместо пользователя."""
import asyncio
import sys

import mcp.types as types
from mcp import StdioServerParameters, stdio_client
from mcp.client.session import ClientSession


async def on_elicit(context, params: types.ElicitRequestParams) -> types.ElicitResult:
    print(f"[клиент] пришёл запрос подтверждения: {params.message}", flush=True)
    return types.ElicitResult(action="accept", content={"approve": True})


async def main() -> None:
    params = StdioServerParameters(command=sys.executable, args=["tools/probes/p3_launcher.py"])
    async with stdio_client(params) as (read, write):
        async with ClientSession(read, write, elicitation_callback=on_elicit) as session:
            await session.initialize()
            tools = await session.list_tools()
            print("[клиент] тулы:", [t.name for t in tools.tools], flush=True)
            result = await session.call_tool("odata1c_probe_commit", {"pending_id": "p-1"})
            print("[клиент] результат тула:", result.content, flush=True)


if __name__ == "__main__":
    asyncio.run(main())
```

- [ ] **Шаг 2: Написать пробу с гипотезой алгоритма**

Создать `tools/probes/p5_account_key.py`. Алгоритм ниже — гипотеза по описанию ЦБ, её и проверяет
шаг 3; итоговые веса и правило префикса берутся из результата, а не из этого файла:

```python
"""Проба P5 (SPEC §15.4): проверка контрольного ключа счёта по БИК.

Гипотеза: к 20 цифрам счёта слева приписывается трёхзначный префикс, зависящий от вида счёта;
получившиеся 23 цифры взвешиваются циклом 7-1-3; счёт верен, если сумма произведений,
взятых по модулю 10, кратна 10.
"""
WEIGHTS = (7, 1, 3)


def prefix_for(account: str, bic: str) -> str:
    """Префикс: для счетов в подразделении Банка России — 0 и две последние цифры БИК,
    для остальных — три последние цифры условного номера банка (разряды 7-9 БИК)."""
    if account.startswith(("30101", "30125")):
        return "0" + bic[-2:]
    return bic[6:9]


def account_key_valid(account: str, bic: str) -> bool:
    digits = prefix_for(account, bic) + account
    if len(digits) != 23 or not digits.isdigit():
        return False
    total = sum((int(d) * WEIGHTS[i % 3]) % 10 for i, d in enumerate(digits))
    return total % 10 == 0


PAIRS: list[tuple[str, str, bool]] = [
    # (БИК, счёт, ожидание) — заполняется на шаге 1 из собранных пар,
    # плюс заведомо неверные варианты: та же пара с испорченной девятой цифрой счёта.
]

if __name__ == "__main__":
    failures = 0
    for bic, account, expected in PAIRS:
        got = account_key_valid(account, bic)
        mark = "ок" if got == expected else "РАСХОЖДЕНИЕ"
        if got != expected:
            failures += 1
        print(f"{mark}: БИК {bic} счёт {account} ожидалось {expected}, получено {got}")
    print(f"расхождений: {failures} из {len(PAIRS)}")
```

### Задача 5: P6 — скорость стража при словаре в 100 000 значений

Отвечает на SPEC §15 пункт 6. Страж — последний проход по каждому ответу (SPEC §6.8), то есть его
стоимость платится на каждом вызове любого тула. Если проход или пересборка автомата дороги, схему
«инкрементальная пересборка при добавлении в словарь» придётся менять.

**Файлы:**
- Создать: `tools/probes/p6_guard_bench.py`
- Создать: `docs/probes/P6-guard-performance.md`

**Интерфейсы:**
- Потребляет: `ahocorasick_rs` из окружения пробы P1.
- Отдаёт: замеры, на которые опирается решение о пересборке автомата в плане M1c (задача «страж»).

- [ ] **Шаг 1: Написать бенчмарк**

Создать `tools/probes/p6_guard_bench.py`:

```python
"""Проба P6 (SPEC §15.6): стоимость прохода стража и пересборки автомата.

Замеряет: сборку автомата на 100 000 вариантов названий, поиск в ответе размером 120 000 символов
(предел result_chars из SPEC §3.3), пересборку после добавления одного значения и проверку
цифровых последовательностей по множеству нормализованных значений словаря.
"""
import random
import re
import string
import time

from ahocorasick_rs import AhoCorasick, MatchKind

DICT_SIZE = 100_000
RESPONSE_CHARS = 120_000
DIGITS_RE = re.compile(r"\d[\d\s\-]{7,}\d")


def make_names(n: int) -> list[str]:
    random.seed(20260907)
    parts = ["ромашка", "вектор", "альфа", "строй", "торг", "сервис", "групп", "инвест"]
    return [f"{random.choice(parts)}-{random.choice(parts)}-{i}" for i in range(n)]


def make_numbers(n: int) -> set[str]:
    random.seed(20260908)
    return {"".join(random.choices(string.digits, k=random.choice((10, 12, 20)))) for _ in range(n)}


def make_response(names: list[str], numbers: set[str]) -> str:
    """Ответ с редкими вхождениями: типичный случай — страж ничего не находит."""
    random.seed(20260909)
    filler = "оплата по счёту от 2026-09-07 на сумму 145 200,00 руб. без НДС; "
    text = filler * (RESPONSE_CHARS // len(filler))
    return text + random.choice(names) + " " + random.choice(list(numbers))


def bench(label: str, fn, repeats: int = 5) -> float:
    best = min(_timed(fn) for _ in range(repeats))
    print(f"{label}: {best * 1000:.1f} мс")
    return best


def _timed(fn) -> float:
    start = time.perf_counter()
    fn()
    return time.perf_counter() - start


def main() -> None:
    names = make_names(DICT_SIZE)
    numbers = make_numbers(DICT_SIZE)
    response = make_response(names, numbers)
    print(f"словарь: {len(names)} названий, {len(numbers)} значений; ответ: {len(response)} символов")

    automaton: AhoCorasick | None = None

    def build() -> None:
        nonlocal automaton
        automaton = AhoCorasick(names, matchkind=MatchKind.LeftmostLongest)

    bench("сборка автомата", build)
    bench("поиск названий в ответе", lambda: automaton.find_matches_as_indexes(response.lower()))
    bench("пересборка после добавления одного названия",
          lambda: AhoCorasick(names + ["новое-название-1"], matchkind=MatchKind.LeftmostLongest))

    def scan_numbers() -> None:
        for match in DIGITS_RE.finditer(response):
            normalized = re.sub(r"\D", "", match.group())
            if normalized in numbers:
                pass

    bench("сверка цифровых последовательностей", scan_numbers)


if __name__ == "__main__":
    main()
```
