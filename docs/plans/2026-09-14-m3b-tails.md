def test_plugin_json_канонический_и_bump_меняет_одну_строку() -> None:
    """м-9 итогового ревью M3: манифест хранится ровно в той форме, в какой его пишет
    `bump_version._записать_json`. Иначе первый же выпуск перекладывает весь файл, и разница
    коммита выпуска не читается глазами.

    Тест ничего не переписывает: он сравнивает файл с его же канонической сериализацией и
    отдельно считает, сколько строк изменила бы подмена версии. `strict=True` у `zip` заодно
    ловит изменение ЧИСЛА строк."""
    путь = КОРЕНЬ / "plugin/.claude-plugin/plugin.json"
    текст = путь.read_text(encoding="utf-8")
    данные = json.loads(текст)
    assert текст == json.dumps(данные, ensure_ascii=False, indent=2) + "\n"

    данные["version"] = "9.9.9"
    новый = json.dumps(данные, ensure_ascii=False, indent=2) + "\n"
    различия = [
        (было, стало)
        for было, стало in zip(текст.splitlines(), новый.splitlines(), strict=True)
        if было != стало
    ]
    assert len(различия) == 1
```

СУЩНОСТЬ = "Catalog_БанковскиеСчета"
ПОЛЕ_КЛАССА = "Владелец/Description"   # название лежит в поле с классом
ПОЛЕ_ТЕКСТА = "Наименование"           # и оно же — внутри свободного текста строки


async def раздел_два_прохода(вызов) -> list[dict]:
    """Один запрос к сущности, где название лежит и в поле с классом, и в свободном тексте.

    Находка засчитывается по ФОРМЕ, а не по значению (инвариант 1, Global Constraints): из
    ответа берётся только признак «в свободном тексте остался открытый фрагмент, который в
    соседнем поле закрыт токеном» и форма этого фрагмента (`форма()`: цифра → 9, буква → a).
    """
    ответ = await вызов(
        "odata1c_query",
        {
            "base": БАЗА,
            "entity": СУЩНОСТЬ,
            "select": [ПОЛЕ_КЛАССА, ПОЛЕ_ТЕКСТА],
            "expand": ["Владелец"],
            "top": 50,
        },
    )
    находки = []
    for запись in json.loads(ответ)["items"]:
        текст = запись.get(ПОЛЕ_ТЕКСТА) or ""
        токен = (запись.get("Владелец") or {}).get("Description") or ""
        if not parse_token(токен):
            continue
        открытые = [
            кусок
            for кусок in _слова(текст)
            if len(кусок) >= 4 and not parse_token(кусок) and _похоже_на_название(кусок)
        ]
        if открытые:
            находки.append({"поле": ПОЛЕ_ТЕКСТА, "формы": sorted({форма(к) for к in открытые})})
    return находки
```

или в кавычках длиной ≥ 4; `_слова` — разбиение по небуквенно-цифровым символам с сохранением
кавычек. Ключ `--label до|после` пишет отчёт с этой пометкой.

        # названий, а автомат известных названий — их варианты. Второй (ниже, обычный) обходит

        # выдал первый. Результат первого прохода выбрасывается: единственное, ради чего он
        # делается, — состояние словаря; поля с классом второй проход токенизирует заново, и
        # `Dictionary.token_for` на известном значении возвращает тот же токен, не двигая
        # `revision()`.

        # ранним проходом (Ruling 25/32), первый проход пропускает без изменений; её настоящее

        # `guard_replaced`.

        # `persist=False` (тул `odata1c_journal`, Ruling 58) первого прохода не получает вовсе:
        # словарь при нём не пополняется, и учить автомат нечему.
        if persist:

                entity=entity,

# M-7 (хвост M1d): пара `"odata.count": "<цифры>"` верхнего уровня ответа. Три шаблона по
# порядку, а не один с необязательными запятыми с обеих сторон: жадная форма
# `(?:,\s*)?…(?:\s*,)?` на счётчике ПОСРЕДИ объекта съедает обе запятые и оставляет невалидный
# JSON. Первый шаблон забирает запятую справа, второй — слева, третий — счётчик как
# единственный ключ. Кавычки вокруг цифр необязательны: JSON OData v3 отдаёт счётчик строкой
# (проба P4), но числовая форма ранним проходом задевалась бы так же.
_СЧЁТЧИК_ШАБЛОНЫ = (
    re.compile(r'"odata\.count"\s*:\s*"?(\d+)"?\s*,'),
    re.compile(r',\s*"odata\.count"\s*:\s*"?(\d+)"?'),
    re.compile(r'"odata\.count"\s*:\s*"?(\d+)"?'),
)


def _вынуть_счётчик(text: str) -> tuple[str, int | None]:
    """Текст ответа без пары `odata.count` и её значение числом (`None` — пары нет).

    Счётчик — служебное поле шлюза (SPEC §5.1, `response.strip_service`), а не данные 1С: ранний
    проход по сырому телу (Ruling 25) не должен видеть его вовсе. Возвращается он уже в
    РАЗОБРАННЫЙ ответ, а не в текст: тогда смещения замен, которые ранний проход запоминает для
    `load_scrubbed` (Ruling 32), считаются по тому же тексту, который разбирается."""
    for шаблон in _СЧЁТЧИК_ШАБЛОНЫ:
        совпадение = шаблон.search(text)
        if совпадение is not None:
            без_счётчика = text[: совпадение.start()] + text[совпадение.end() :]
            return без_счётчика, int(совпадение.group(1))
    return text, None

## Самопроверка плана

Проведена после написания; найденное исправлено прямо в тексте выше.

1. **Покрытие.** Каждый раздел проекта закрыт задачей — таблица выше. Разделов без задачи нет.
2. **Заглушек нет.** Развилки по отчёту P9 не оставлены на усмотрение исполнителя формулой
   «сузить бриф»: обе ветки P9-6 и P9-8 написаны конкретным кодом и конкретным тестом
   (задача 6 шаги 6 и 9; ветки P9-1/P9-3 ушли вместе с задачей 7, Ruling 100). Нигде не сказано «как в задаче N» — код
   повторяется там, где нужен. Два места, где сначала стояло «сюда переносится существующий
   код», переписаны: разделяемая перестройка сведена к одному вызову `rebuild_index` (задача 4,
   шаг 4), а `остановить` выписана целиком, со всеми пятью исходами (задача 4, шаг 9).
3. **Согласованность имён.** Сверено между задачами: `rows_diff`/`RowsDiff` (задача 6 →
   задача 9, раздел `часть_update`); `PendingStore.count_live` и код `pending_limit` (6 → 9,
   раздел `предел`); `Recipe.expand` (8 → 9, раздел
   `рецепты`); `add_custom_class`/`copy_policy` (12); `password_missing` и
   `secrets.ЛИТЕРАЛ` (10 → 11, навык); `GET /alive` с полем `pid` (4 → 9, раздел `стоп`);
   (`Посредник1С(..., удалять=…)` снят вместе с регистрами, Ruling 100).
4. **Отклонения от буквы проекта — названы, а не сделаны молча.**
   - §3.8 просит `asyncio.Lock` вокруг `reindex` в `daemon.py`. В плане — словарь текущих задач
     в `ToolService.reindex`, и разделяется ровно `rebuild_index` (задача 4, шаг 4). Причина:
     обычный замок заставил бы второй вызов ПОВТОРИТЬ перестройку, а проект требует, чтобы он
     «получил результат первого»; и оба вызывающих (тул демона и фоновая проверка `$metadata`)
     идут через `ToolService.reindex`, а не через `daemon.py` напрямую. Тест проверяет именно
     это: `$metadata` скачан один раз. Пересборка политики и стража осталась у каждого
     вызывающего — она идемпотентна, и переносить её в общий метод значило бы вынимать из
     замыкания шесть десятков строк ради нуля выигрыша.
   - §8 описывает три исхода `stop()`. В плане их четыре (задача 4, шаг 9): добавлен «порт
     отвечает, но номера не назвал» — демон версии до M3b. Без него `stop` объявил бы pid-файл
     осиротевшим ровно в том случае, ради которого его чаще всего и зовут: остановить старый
     демон, чтобы поднять новый код.
   - §5.4 говорит об отпечатке «полное содержимое записи»; `CONTEXT.md` (термин «Отпечаток») —
     «SHA-256 полей записи регистра». В плане взята формулировка `CONTEXT.md` с канонизацией
     (задача 7, шаг 7, пункт 2): расхождение между документами — дефект, а словарь проекта
     главнее.
   - §6.1 просит предупреждение у базы без секрета в `odata1c_bases`. В плане это ОТДЕЛЬНАЯ
     проверка внутри метода `bases` (задача 10, шаг 5): перечень баз клиента не создаёт вовсе, и
     без своей проверки предупреждение не появилось бы никогда.
